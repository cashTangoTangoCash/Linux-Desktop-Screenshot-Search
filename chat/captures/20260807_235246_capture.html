#!/usr/bin/env python3 import os import re import shutil from pathlib import Path from concurrent.futures import ProcessPoolExecutor, as_completed import pytesseract from PIL import Image

WORK_DIR = Path.cwd()

Cap workers to prevent fans from maxing out (e.g., 2 to 4 cores)

MAX_WORKERS = 6

RULES = [     (r"craigslist.org", "craigslist"),     (r"wikipedia.org", "wikipedia"),     (r"howard\shanna|mls\s#?\s*[a-z]?\d{6,8}|new\s*listing", "housing-listings"),     (r"news.google.com", "google-news"),     (r"cambriabike.com", "shopping-cambria-bike"),     (r"jensonusa.com", "shopping-jenson-usa-bike"),     (r"amazon.com", "shopping-amazon"),     (r"bikeradar.com", "shopping-bikeradar-bike"),     (r"rottentomatoes.com", "movies"),     (r"audioclassics.com", "audioclassics"),     (r"forecast.weather.gov", "weather"),     (r"npr.org", "news"),     (r"ebay.com", "shopping-ebay"),     (r"github.com", "github"),     (r"arstechnica.com", "technology"),     (r"gizmodo.com", "news"),     (r"nytimes.com", "news"),     (r"theguardian.com", "news"),     (r"techcrunch.com", "technology"), ]

def process_single_image(image_path: Path):     """Worker function: reads cached .txt or runs OCR to generate sidecar file."""     txt_path = WORK_DIR / f"{image_path.name}.txt"

    if txt_path.is_file():         try:             extracted_text = txt_path.read_text(encoding="utf-8")         except Exception:             extracted_text = ""     else:         try:             img = Image.open(image_path)             extracted_text = pytesseract.image_to_string(img).lower()             txt_path.write_text(extracted_text, encoding="utf-8")         except Exception as e:             extracted_text = ""

    return image_path, txt_path, extracted_text

def batch_sort():     image_extensions = {".png", ".jpg", ".jpeg", ".webp"}     image_files = [         f for f in WORK_DIR.iterdir()         if f.is_file() and f.suffix.lower() in image_extensions     ]

    if not image_files:         print(f"No unprocessed images found in {WORK_DIR}")         return

    print(f"Processing {len(image_files)} image(s) using up to {MAX_WORKERS} workers...\n")

    # Run OCR across process pool     results = []     with ProcessPoolExecutor(max_workers=MAX_WORKERS) as executor:         futures = [executor.submit(process_single_image, img_path) for img_path in image_files]         for future in as_completed(futures):             results.append(future.result())

    # Sort files based on rules     for image_path, txt_path, extracted_text in results:         subfolder_name = None         for pattern, folder in RULES:             if re.search(pattern, extracted_text):                 subfolder_name = folder                 break

        if subfolder_name:             target_dir = WORK_DIR / subfolder_name             target_dir.mkdir(parents=True, exist_ok=True)

            shutil.move(str(image_path), str(target_dir / image_path.name))

            if txt_path.is_file():                 shutil.move(str(txt_path), str(target_dir / txt_path.name))

            print(f"Moved '{image_path.name}' -> {subfolder_name}/")         else:             print(f"Skipped '{image_path.name}' (no rule matched)")

if name == "main":     batch_sort()
